34template <typename Alloc, typename T>
35auto make_with_allocator(const Alloc& alloc,
36 T&& obj) noexcept(!::std::uses_allocator_v<::std::remove_cvref_t<T>, Alloc> &&
37 ::std::is_nothrow_constructible_v<::std::remove_cvref_t<T>, T>)
38 -> ::std::remove_cvref_t<T> {
39 if constexpr (::std::uses_allocator_v<::std::remove_cvref_t<T>, Alloc>)
40 return ::std::make_obj_using_allocator<::std::remove_cvref_t<T>>(alloc, ::std::forward<T>(obj));
41 else
42 return ::std::remove_cvref_t<T>(::std::forward<T>(obj));
43}
44
45template <typename Alloc, typename T>
46 requires(decltype(::beman::execution::detail::is_product_type(::std::declval<const T&>()))::value)
47auto make_with_allocator(const Alloc& alloc, T&& obj) noexcept(
48 noexcept(::std::remove_cvref_t<T>::make_from(alloc, ::std::forward<T>(obj)))) -> ::std::remove_cvref_t<T> {
49 return ::std::remove_cvref_t<T>::make_from(alloc, ::std::forward<T>(obj));
50}
51
60template <typename T, typename Context>
61auto allocator_aware_move(T&& obj, Context&&) noexcept -> T&& {
62 return ::std::forward<T>(obj);
63}
64
65template <typename T, typename Context>
66 requires ::std::invocable<::beman::execution::get_allocator_t, ::beman::execution::env_of_t<Context>>
67auto allocator_aware_move(T&& obj, Context&& context) noexcept(
68 noexcept(::beman::execution::detail::make_with_allocator(
69 ::beman::execution::get_allocator(::beman::execution::get_env(context)), ::std::forward<T>(obj))))
70 -> ::std::remove_cvref_t<T> {
71 return ::beman::execution::detail::make_with_allocator(
72 ::beman::execution::get_allocator(::beman::execution::get_env(context)), ::std::forward<T>(obj));
73}
