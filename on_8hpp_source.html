73struct on_t : ::beman::execution::sender_adaptor_closure<on_t> {
74 template <typename Env, typename OutSndr>
75 static auto dry_run(OutSndr&& out_sndr) noexcept {
76 auto& [_, data, child] = out_sndr;
77 if constexpr (::beman::execution::scheduler<decltype(data)>) {
78 return ::std::bool_constant<::std::invocable<::beman::execution::get_start_scheduler_t, Env>>{};
79 } else {
80 ::std::bool_constant<
81 ::std::invocable<::beman::execution::get_completion_scheduler_t<::beman::execution::set_value_t>,
82 ::beman::execution::env_of_t<decltype(child)>,
83 Env>>{};
84 }
85 }
86
87 template <::beman::execution::detail::sender_for<on_t> OutSndr, typename Env>
88 requires(decltype(on_t::dry_run<Env>(::std::declval<OutSndr>()))::value)
89 static auto transform_sender(::beman::execution::set_value_t, OutSndr&& out_sndr, const Env& env)
90 -> decltype(auto) {
91 auto&& data = out_sndr.template get<1>();
92 auto&& child = out_sndr.template get<2>();
93
94 if constexpr (::beman::execution::scheduler<decltype(data)>) {
95 auto start_sch = ::beman::execution::get_start_scheduler(env);
96 return ::beman::execution::continues_on(
97 ::beman::execution::starts_on(::beman::execution::detail::forward_like<OutSndr>(data),
98 ::beman::execution::detail::forward_like<OutSndr>(child)),
99 ::std::move(start_sch));
100 } else {
101 auto& [sch, closure] = data;
102 auto orig_sch = ::beman::execution::get_completion_scheduler<::beman::execution::set_value_t>(
103 ::beman::execution::get_env(child), env);
104
105 return ::beman::execution::continues_on(
106 ::beman::execution::detail::forward_like<OutSndr>(closure)(
107 ::beman::execution::continues_on(::beman::execution::detail::forward_like<OutSndr>(child), sch)),
108 orig_sch);
109 }
110 }
